async function init() { const user = "simonlukas102030"; const repo = "Quick-share"; const list = document.getElementById('file-list'); const hashOutput = document.getElementById('generated-hash'); const lastInputDisplay = document.getElementById('last-input'); try { const pRes = await fetch(`https://raw.githubusercontent.com/${user}/${repo}/main/passwords.json?t=${Date.now()}`); if(pRes.ok) passwords = await pRes.json(); const res = await fetch(`https://api.github.com/repos/${user}/${repo}/contents/`); const files = await res.json(); list.innerHTML = ""; files.forEach(file => { if (file.name.endsWith('.html') && file.name.toLowerCase() !== 'index.html') { const li = document.createElement('li'); li.className = 'file-item'; // Prüfen, ob für diese Datei ein Passwort existiert const hasPassword = passwords.hasOwnProperty(file.name); const a = document.createElement('a'); a.textContent = file.name; if(hasPassword) a.classList.add('locked'); a.onclick = async () => { // NUR fragen, wenn ein Passwort hinterlegt ist if (hasPassword) { const pw = prompt("Passwort erforderlich für " + file.name + ":"); if (!pw) return; const hashedInput = await sha256(pw); // Debug-Leiste unten aktualisieren lastInputDisplay.textContent = pw; hashOutput.textContent = hashedInput; if (hashedInput === passwords[file.name].toLowerCase().trim()) { window.location.href = file.name; } else { alert("Falsch! Siehe Hash unten."); } } else { // Wenn kein Passwort da ist, sofort weiterleiten window.location.href = file.name; } }; li.appendChild(a); list.appendChild(li); } }); } catch (e) { list.innerHTML = "
